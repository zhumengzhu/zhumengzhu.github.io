JIT deoptimization 风暴导致 CPU 使用率飙升

JIT deoptimization 风暴导致 CPU 使用率飙升，本文尝试分析其原因，并给出了一个测试代码，可以复现该问题。

2023/09/23
最后修改于: 2024/12/20 14:14 CST
阅读时长: 2 分钟

背景

在对某抽奖压测服务时(dev&test环境)，发现当 QPS 达到 100 左右时，CPU 使用率就达到 100%，对应的 on-CPU 火焰图显示，有接近 56% 的 CPU 时间在进行 Deoptimization：

经调研，怀疑是该 JDK Bug 导致：https://bugs.openjdk.org/browse/JDK-8243615 类似的反馈还有：

  • https://mail.openjdk.org/pipermail/hotspot-dev/2016-June/023316.html
  • https://mail.openjdk.org/pipermail/hotspot-compiler-dev/2016-June/023278.html
  • https://github.com/trinodb/trino/issues/6405
  • https://trino.io/blog/2021/10/06/jvm-issues-at-comcast.html#jit-recompilation

public class UnstableIfTest {
    static int compute(double x)
    {
        return (int) (Math.sin(x) + Math.cos(x));
    }
    static void hotMethod(int iteration)
    {
        if (iteration < 20) {
            compute(78.3);
        }
        else if (iteration < 40) {
            compute(78.3);
        }
        else if (iteration < 60) {
            compute(78.3);
        }
        else if (iteration < 80) {
            compute(78.3);
        } else if (iteration < 100) {
            compute(78.3);
        } else {
            compute(78.3);
        }
    }
    static void hotMethodWrapper(int iteration)
    {
        int count = 100_000;
        for (int i = 0; i < count; i++) {
            hotMethod(iteration);
        }
    }
    public static void main(String[] args)
    {
        for (int k = 0; k < 1000; k++) {
            long start = System.nanoTime();
            hotMethodWrapper(k + 1);
            System.out.println("iteration " + k + ": " + (System.nanoTime() - start) / 1_000_000 + "ms");
        }
    }
}
